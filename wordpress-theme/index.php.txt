<?php
/**
 * Template principal
 * 
 * @package Mentoria_FULL
 */

get_header();
?>

<main id="primary" class="site-main">
    
    <?php
    if (have_posts()) :
        while (have_posts()) :
            the_post();
            ?>
            
            <article id="post-<?php the_ID(); ?>" <?php post_class(); ?>>
                <div class="container" style="padding: 80px 20px;">
                    
                    <header class="entry-header" style="text-align: center; margin-bottom: 40px;">
                        <?php the_title('<h1 class="entry-title section-title">', '</h1>'); ?>
                        <div class="entry-meta" style="color: #b0b0b0; margin-top: 12px;">
                            <time datetime="<?php echo get_the_date('c'); ?>">
                                <?php echo get_the_date(); ?>
                            </time>
                            <?php if (get_the_author()) : ?>
                                <span> | Por <?php the_author(); ?></span>
                            <?php endif; ?>
                        </div>
                    </header>
                    
                    <?php if (has_post_thumbnail()) : ?>
                        <div class="entry-thumbnail" style="margin-bottom: 40px; border-radius: 24px; overflow: hidden;">
                            <?php the_post_thumbnail('large', array('style' => 'width: 100%; height: auto;')); ?>
                        </div>
                    <?php endif; ?>
                    
                    <div class="entry-content" style="max-width: 800px; margin: 0 auto; line-height: 1.8; font-size: 1.125rem;">
                        <?php
                        the_content();
                        
                        wp_link_pages(array(
                            'before' => '<div class="page-links">' . esc_html__('Páginas:', 'mentoria-full'),
                            'after'  => '</div>',
                        ));
                        ?>
                    </div>
                    
                    <footer class="entry-footer" style="margin-top: 60px; padding-top: 40px; border-top: 2px solid rgba(228, 195, 132, 0.2);">
                        <?php
                        if (has_tag()) {
                            echo '<div style="margin-bottom: 20px;">';
                            the_tags('<strong>Tags: </strong>', ', ', '');
                            echo '</div>';
                        }
                        
                        if (has_category()) {
                            echo '<div>';
                            echo '<strong>Categorias: </strong>';
                            the_category(', ');
                            echo '</div>';
                        }
                        ?>
                    </footer>
                    
                </div>
            </article>
            
            <?php
            // Comentários
            if (comments_open() || get_comments_number()) :
                comments_template();
            endif;
            
        endwhile;
        
        // Paginação
        the_posts_navigation();
        
    else :
        ?>
        <div class="container" style="padding: 120px 20px; text-align: center;">
            <h2 class="section-title">Nada Encontrado</h2>
            <p class="section-subtitle">Desculpe, mas nenhum conteúdo foi encontrado. Tente fazer uma busca.</p>
            <?php get_search_form(); ?>
        </div>
        <?php
    endif;
    ?>
    
</main>

<?php
get_footer();
